add_library(polly-execution STATIC thread.c dispatch.c)
target_include_directories(polly-execution PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
if (WIN32)
    target_compile_definitions(polly-execution PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
else()
    find_package(Threads REQUIRED)
    target_compile_definitions(polly-execution PRIVATE _POSIX_C_SOURCE=200809L)
    target_link_libraries(polly-execution PUBLIC Threads::Threads)
endif()
