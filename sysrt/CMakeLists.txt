cmake_minimum_required(VERSION 3.25)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(PollySystemRT LANGUAGES C)
    include(CTest)
    set(CMAKE_C_STANDARD 11)
    set(CMAKE_C_STANDARD_REQUIRED ON)
    set(CMAKE_C_EXTENSIONS OFF)
    if (MSVC)
        set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
    endif()
    set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    set(QJS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(../third_party/quickjs "${CMAKE_CURRENT_BINARY_DIR}/quickjs" EXCLUDE_FROM_ALL)
    add_subdirectory(../shared "${CMAKE_CURRENT_BINARY_DIR}/shared")
endif()
get_filename_component(SR_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

if (TARGET qjs)
    if (WIN32)
        set(SYSRT_LIBFFI_ROOT "${SR_REPOSITORY_ROOT}/build/sysrt-deps/x64-windows-static"
            CACHE PATH "Static-CRT libffi package prefix")
        find_package(unofficial-libffi CONFIG QUIET
            PATHS "${SYSRT_LIBFFI_ROOT}/share/unofficial-libffi")
        if (NOT TARGET unofficial::libffi::libffi)
            message(FATAL_ERROR "SysRT requires static-CRT libffi. Run tools/bootstrap-sysrt.ps1 "
                "or set SYSRT_LIBFFI_ROOT to its package prefix.")
        endif()
        set(SR_FFI_LIBRARY unofficial::libffi::libffi)
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(LIBFFI REQUIRED IMPORTED_TARGET libffi>=3.4)
        set(SR_FFI_LIBRARY PkgConfig::LIBFFI)
    endif()
    add_library(polly-sysrt-ffi STATIC ffi/module.c)
    target_include_directories(polly-sysrt-ffi PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
    target_link_libraries(polly-sysrt-ffi PUBLIC qjs PRIVATE ${SR_FFI_LIBRARY} ${CMAKE_DL_LIBS})
    if (WIN32)
        target_compile_definitions(polly-sysrt-ffi PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
    else()
        target_compile_definitions(polly-sysrt-ffi PRIVATE _POSIX_C_SOURCE=200809L)
        target_compile_options(polly-sysrt-ffi PRIVATE -Wall -Wextra -Wpedantic -Werror)
        target_link_libraries(polly-sysrt-ffi PRIVATE m)
    endif()
    if (BUILD_TESTING)
        add_library(polly-sysrt-ffi-fixture SHARED tests/ffi-fixture.c)
        add_executable(polly-sysrt-ffi-test tests/ffi-runner.c)
        add_executable(polly-sysrt-process-child tests/process-child.c)
        target_link_libraries(polly-sysrt-ffi-test PRIVATE polly-sysrt-ffi)
        if (WIN32)
            target_compile_definitions(polly-sysrt-ffi-test PRIVATE _CRT_SECURE_NO_WARNINGS)
        endif()
        add_test(NAME sysrt-ffi COMMAND polly-sysrt-ffi-test
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/ffi.mjs" $<TARGET_FILE:polly-sysrt-ffi-fixture>)
        set_tests_properties(sysrt-ffi PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
        if (WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
            add_test(NAME sysrt-process COMMAND polly-sysrt-ffi-test
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/process-native.mjs"
                $<TARGET_FILE:polly-sysrt-ffi-fixture> $<TARGET_FILE:polly-sysrt-process-child>)
            set_tests_properties(sysrt-process PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
            add_test(NAME sysrt-network COMMAND polly-sysrt-ffi-test
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/network-native.mjs" $<TARGET_FILE:polly-sysrt-ffi-fixture>)
            set_tests_properties(sysrt-network PROPERTIES TIMEOUT 30 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")
        endif()
        if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
            find_program(SR_PYTHON python3 REQUIRED)
            add_test(NAME sysrt-files COMMAND "${SR_PYTHON}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/files-fixture.py"
                $<TARGET_FILE:polly-sysrt-ffi-test> $<TARGET_FILE:polly-sysrt-ffi-fixture>)
            set_tests_properties(sysrt-files PROPERTIES TIMEOUT 180 LABELS "ordinary-native;private-files-fixture")
        endif()
    endif()
    add_library(polly-sysrt-qjs STATIC projection/quickjs/storage.c
        projection/quickjs/fetch.c providers/app_paths.c)
    target_include_directories(polly-sysrt-qjs PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/..")
    target_link_libraries(polly-sysrt-qjs PUBLIC qjs polly-execution)
    if (WIN32)
        target_compile_definitions(polly-sysrt-qjs PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX)
        target_link_libraries(polly-sysrt-qjs PRIVATE winhttp)
    else()
        target_compile_definitions(polly-sysrt-qjs PRIVATE _POSIX_C_SOURCE=200809L)
    endif()
    if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
        target_link_libraries(polly-sysrt-qjs PRIVATE PkgConfig::CURL)
    endif()
endif()
